เนื้อหานี้มุ่งเน้นการทำความเข้าใจสถาปัตยกรรมภาษา VHDL (Hardware Description Language) ตั้งแต่พื้นฐานโครงสร้าง พฤติกรรมฮาร์ดแวร์ กฎการตั้งชื่อสัญญาณมาตรฐานอุตสาหกรรม จนถึงการออกแบบโมดูลขั้นสูง

1 โครงสร้างภาษา VHDL: Entity & Architecture

โครงสร้างหลักของไฟล์ VHDL แบ่งออกเป็น 3 ส่วนหลักตามลำดับดังแสดงในรูป:

Figure 2-1

  1. Library Declaration (ส่วนประกาศไลบรารี):

    • ทำหน้าที่ระบุไลบรารีและแพ็กเกจที่จะดึงมาใช้งาน เช่น library IEEE; use IEEE.STD_LOGIC_1164.ALL; เพื่อใช้งานชนิดข้อมูลมาตรฐานฮาร์ดแวร์

  2. Entity Declaration (ส่วนประกาศเอนทิตี):

    • เปรียบเสมือน "กรอบกล่องฮาร์ดแวร์ภายนอก" ที่กำหนดชื่อโมดูล พารามิเตอร์ปรับขนาดวงจร (Generic) และสัญญาณขาเข้า/ขาออก (Port)

  3. Architecture Definition (ส่วนนิยามสถาปัตยกรรม):

    • เปรียบเสมือน "วงจรภายในกล่อง" ที่บรรยายการทำงานของฮาร์ดแวร์ โดยแบ่งโครงสร้างภายในออกเป็น 2 โซนสำคัญ:
      • Declarative Region (โซนประกาศก่อนคำสั่ง begin):
        • Component Declaration: ประกาศโมดูลย่อย (Sub-modules) ที่ต้องการดึงมาต่อร่วมในวงจร
        • Signal Declaration: ประกาศสายสัญญาณภายใน (Internal Nodes) และสลักข้อมูล/เรจิสเตอร์ (Registers)
      • Statement Region (โซนบรรยายวงจรหลังคำสั่ง begin):
        • เขียนบรรยายพฤติกรรมวงจร (Circuit Description) ด้วยคำสั่งขนาน (Concurrent Statements), การต่อสายโมดูลย่อย (Component Instantiation), หรือบล็อกการทำงานตามลำดับ (process)
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

-- 1. Library & Entity Declaration (กรอบกล่องภายนอก)
entity my_module is
    Generic (
        G_DATA_WIDTH : integer := 8  -- Generic Parameter
    );
    Port (
        clk   : in  std_logic;
        rst_n : in  std_logic;
        i_data: in  std_logic_vector(G_DATA_WIDTH-1 downto 0);
        o_data: out std_logic_vector(G_DATA_WIDTH-1 downto 0)
    );
end entity my_module;

-- 2. Architecture Definition (วงจรภายในกล่อง)
architecture Behavioral of my_module is
    -- [Declarative Region] ประกาศ Signal / Component ภายใน
    signal s_data_reg : std_logic_vector(G_DATA_WIDTH-1 downto 0);
begin
    -- [Statement Region] บรรยายวงจรฮาร์ดแวร์หลังคำสั่ง begin
    process(clk, rst_n)
    begin
        if rst_n = '0' then
            s_data_reg <= (others => '0');
        elsif rising_edge(clk) then
            s_data_reg <= i_data;
        end if;
    end process;

    o_data <= s_data_reg;
end architecture Behavioral;
ข้อควรระวัง: การกำหนดค่าเริ่มต้นที่ Signal Declaration

การเขียน signal s_data_reg : std_logic_vector(...) := (others => '0'); เป็นการกำหนดค่า Initial Value สำหรับ Simulation เท่านั้น (ค่า ณ เวลา 0 ns) ไม่ใช่วงจร Reset จริง — บน FPGA จริง ค่านี้จะถูกโหลดผ่านกลไก Configuration/Bitstream ของแต่ละเวนเดอร์เท่านั้น (ใช้ได้กับบาง Flow) และไม่สามารถพึ่งพาแทนวงจร Reset (เช่น rst_n) ได้ หากต้องการ Reset ที่ทำงานได้จริงหลัง Power-up หรือระหว่างการทำงาน ต้องออกแบบผ่าน process(clk, rst_n) ตามตัวอย่างข้างต้นเสมอ

2 Combinational Logic vs Sequential Logic

ในภาษา VHDL วงจรฮาร์ดแวร์และการประมวลผลคำสั่งแบ่งออกเป็น 2 ประเภทหลัก:

  1. Combinational Logic (วงจรรวมเชิงผสม — นอก Process หรือใน Combinational Process): โค้ดที่อยู่นอก process (เช่น สัญญาณกำหนดค่าธรรมดา, when-else, with-select) รวมถึง Combinational Process จะถูกประมวลผล ขนานกันตลอดเวลา (Parallel Execution) ไม่ว่าเราจะสลับบรรทัดขึ้นลงอย่างไร วงจรฮาร์ดแวร์ที่สังเคราะห์ได้จะมีผลลัพธ์เหมือนกันทุกประการ

  2. Sequential Logic (วงจรเชิงลำดับ — ใน Clocked Process): โค้ดที่อยู่ภายในบล็อก process(...) ที่ทำงานตามขอบสัญญาณนาฬิกา จะถูกประมวลผล เรียงตามลำดับบรรทัดจากบนลงล่าง (Sequential Execution) เมื่อสัญญาณใน Sensitivity List มีการเปลี่ยนแปลงค่า

💡 ข้อสังเกต Combinational Logic:

ถึงแม้โค้ดใน process จะอ่านและประมวลผลเรียงบรรทัด แต่เมื่อนำไปสังเคราะห์เป็นวงจร Combinational Logic (ที่ใส่ Sensitivity List ครบถ้วน) ผลลัพธ์ฮาร์ดแวร์ที่ได้ในชิปจริงก็คือ การทำงานขนานกันตลอดเวลา (Parallel Hardware) ไม่ต่างจากการเขียน Combinational Logic นอก Process

2.1 ตัวอย่างโค้ด VHDL: Combinational Process vs Clocked Sequential Process

ในทางปฏิบัติ การใช้งาน Process แบ่งออกเป็น 2 ประเภทหลัก:

  1. Combinational Process (ไม่มี Clock): ทำงานเมื่ออินพุตเปลี่ยนค่า (ต้องใส่สัญญาณอินพุตทุกตัวใน Sensitivity List หรือใช้ process(all) เพื่อให้ได้วงจร Combinational Logic ขนานที่สมบูรณ์และป้องกัน Latch)
  2. Clocked Sequential Process (มี Clock — ⭐ วิธีมาตรฐานที่แนะนำสำหรับ FPGA): ทำงานเฉพาะที่ขอบสัญญาณนาฬิกา (rising_edge(clk)) สังเคราะห์ได้ Flip-Flops (Sequential Logic) ที่มีไทม์มิ่งเสถียร ป้องกันการเกิด Glitch และ Latch โดยสมบูรณ์
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity process_types_demo is
    Port (
        clk        : in  std_logic;
        rst_n      : in  std_logic;
        i_a        : in  std_logic;
        i_b        : in  std_logic;
        o_comb_y   : out std_logic; -- ผลลัพธ์จาก Combinational Process
        o_clocked_q: out std_logic  -- ผลลัพธ์จาก Clocked Sequential Process (แนะนำ)
    );
end entity process_types_demo;

architecture Behavioral of process_types_demo is
    signal s_comb_node : std_logic;
begin

    ------------------------------------------------------------------
    -- 1. Combinational Process (ประมวลผลทันทีที่อินพุตเปลี่ยน)
    -- ⚠️ ต้องใส่ "ทุกสัญญาณที่ถูกอ่านค่าใน Process" เข้า Sensitivity List ให้ครบ
    ------------------------------------------------------------------
    process(i_a, i_b, s_comb_node)
    begin
        s_comb_node <= i_a and i_b;
        o_comb_y    <= s_comb_node or i_a;
    end process;

    ------------------------------------------------------------------
    -- 2. Clocked Sequential Process (⭐ แนะนำที่สุดสำหรับการออกแบบ FPGA)
    -- ทำงานเรียงลำดับเฉพาะจังหวะขอบขาขึ้นของ clk (และ Async Reset)
    ------------------------------------------------------------------
    process(clk, rst_n)
    begin
        if rst_n = '0' then
            o_clocked_q <= '0';
        elsif rising_edge(clk) then
            -- สังเคราะห์กลายเป็น D Flip-Flop บันทึกค่าตามจังหวะ Clock
            o_clocked_q <= i_a and i_b;
        end if;
    end process;

end architecture Behavioral;
ข้อสังเกตฮาร์ดแวร์:
  • Combinational Logic: เหมาะสำหรับเขียนสมการ Combinational สั้นๆ (เช่น o_y <= i_a and i_b;, when-else, with-select)
  • Sequential Logic: จำเป็นต้องใช้เมื่อสร้าง Sequential Logic (Flip-Flops / Clock Edge) หรือต้องการใช้โครงสร้างเงื่อนไขที่ซับซ้อน เช่น if-then-else และ case-when

2.2 เจาะลึก Sensitivity List: การทำงานและข้อควรระวัง

Sensitivity List คือรายการสัญญาณที่ระบุไว้ในวงเล็บหลังคำสั่ง process(...) เพื่อบอก Simulator และ Synthesis Tool ว่า "ให้บล็อก Process นี้ประมวลผลใหม่ทุกครั้งที่มีสัญญาณตัวใดตัวหนึ่งในรายการเปลี่ยนค่า (Signal Event)"

2.2.1 Sensitivity List ใน Combinational Process

  • กฎเหล็ก: ต้องใส่ สัญญาณอินพุตทุกตัว ที่ถูกอ่านค่าภายใน Process เข้าไปใน Sensitivity List ให้ครบถ้วน
  • หากใส่ไม่ครบ: จะเกิดปัญหา Simulation vs Synthesis Mismatch (ผลการจำลองใน Simulation ไม่ตรงกับฮาร์ดแวร์จริง) และ Synthesis Tool จะสร้าง Unintentional Latch ขึ้นมาเก็บค่าสัญญาณที่ไม่ได้ถูกอัปเดต

2.2.2 Sensitivity List ใน Clocked Sequential Process

  • กฎเหล็ก: ใส่เฉพาะสัญญาณ Clock (และ Asynchronous Reset ถ้ามี) เท่านั้น!
  • เหตุผล: วงจร Register / Flip-Flop จะตอบสนองและเปลี่ยนสถานะเฉพาะที่ขอบสัญญาณนาฬิกา (rising_edge(clk)) สัญญาณ Data Input อื่นๆ ไม่จำเป็นต้องใส่ใน Sensitivity List เพราะค่าจะถูกจับบันทึก (Sample) ตามจังหวะ Clock เท่านั้น

2.2.3 VHDL-2008 Standard Option: process(all)

ในมาตรฐาน VHDL-2008 เพิ่มฟีเจอร์ process(all) สำหรับ Combinational Logic ช่วยให้ Compiler รวบรวมสัญญาณอินพุตทั้งหมดที่อ่านใน Process เข้า Sensitivity List ให้อัตโนมัติ ช่วยลดข้อผิดพลาดในการลืมใส่สัญญาณให้ครบถ้วน

3 ชนิดข้อมูลและแอตทริบิวต์ (Data Types & Attributes)

3.1 ชนิดข้อมูลหลักใน VHDL (std_logic_1164):

  • std_logic: สัญญาณ 1 บิต มี 9 สถานะ (สถานะหลักคือ '0', '1', 'Z' (High Impedance), 'X' (Unknown))

  • std_logic_vector: อาเรย์ชุดสัญญาณบิต เช่น std_logic_vector(7 downto 0)

  • array: การจำลองหน่วยความจำ เช่น อาเรย์ของ vector:

    type ram_type is array (0 to 255) of std_logic_vector(7 downto 0);
signal s_ram : ram_type;

  • record: กลุ่มข้อมูลรวมหลายประเภทไว้ในชื่อเดียว (คล้าย struct ในภาษา C):

    type bus_header_t is record
    addr : std_logic_vector(15 downto 0);
    data : std_logic_vector(31 downto 0);
    valid: std_logic;
end record;

3.2 แอตทริบิวต์ที่ใช้บ่อย (VHDL Attributes):

  • 'high / 'low: คืนค่าดัชนีสูงสุด / ต่ำสุดของเวกเตอร์
  • 'range / 'reverse_range: คืนค่าช่วงดัชนี เช่น (7 downto 0)
  • 'length: คืนค่าจำนวนบิตทั้งหมดของเวกเตอร์

4 โครงสร้างควบคุมการทำงาน (Control Flow & Hardware Generation)

4.1 คำสั่งควบคุมลำดับภายใน Process (Sequential Control Statements)

  • if-then-else:
    ใช้สร้างเงื่อนไขตามลำดับความสำคัญ (Priority Logic) ซึ่งหากไม่ระบุเงื่อนไข else ครบถ้วนในวงจร Combinational Logic จะทำให้เกิด Unintentional Latch ได้ 
    process(i_enable, i_data)
begin
    if i_enable = '1' then
        o_out <= i_data;
    else
        o_out <= (others => '0');
    end if;
end process;
  • case-when:
    ใช้สร้าง Multiplexer หรือ FSM State Transition โดยบังคับให้ระบุเงื่อนไขให้ครอบคลุมทุกค่าอินพุตที่เป็นไปได้ด้วย when others 
    process(s_state)
begin
    case s_state is
        when "00"   => o_led <= "0001";
        when "01"   => o_led <= "0010";
        when "10"   => o_led <= "0100";
        when others => o_led <= "1000"; -- ป้องกัน Latch
    end case;
end process;
  • for-loop:
    ใช้คลี่สร้างโครงสร้างลอจิกขนานตามจำนวนรอบแบบตายตัว (Unrolling Loop) ในการสังเคราะห์ฮาร์ดแวร์ เช่น การสลับกลับบิตข้อมูล (Bit Reversal): 
    process(i_data)
begin
    for i in 0 to 7 loop
        o_reversed(i) <= i_data(7 - i);
    end loop;
end process;

4.2 คำสั่งสร้างฮาร์ดแวร์ขนานภายนอก Process (Concurrent Generate Statement)

  • for-generate: ถอดแบบโมดูลหรือวงจรขนานตามจำนวนที่กำหนด เช่น สร้าง Buffer 4 ช่อง:

    g_buf_array: for i in 0 to 3 generate
    u_buf : entity work.buffer_cell
        port map (
            i_in  => i_data(i),
            o_out => o_data(i)
        );
end generate g_buf_array;

  • if-generate: เลือกสร้างโครงสร้างฮาร์ดแวร์ตามเงื่อนไข Generic (Conditional Instantiation) เช่น เลือกสร้างวงจร Pipe-lined Register หรือต่อตรงตามค่า Generic G_INCLUDE_REG:

    -- สร้างวงจรเก็บข้อมูลเมื่อ Generic G_INCLUDE_REG = true
g_reg_gen: if G_INCLUDE_REG generate
    u_reg : entity work.data_register
        port map (
            clk   => clk,
            i_d   => i_data,
            o_q   => o_data
        );
end generate g_reg_gen;

-- สร้างวงจรต่อสายตรงเมื่อ Generic G_INCLUDE_REG = false
g_pass_gen: if not G_INCLUDE_REG generate
    o_data <= i_data;
end generate g_pass_gen;

5 การออกแบบเชิงโมดูลขั้นสูง (Package, Functions & Procedures)

5.1 Package (*_pkg.vhd)

ใช้เก็บประกาศประเภทข้อมูลชนิดพิเศษ (type), ค่าคงที่ (constant), และฟังก์ชันส่วนกลาง เพื่อให้ทุกโมดูลในโปรเจกต์อ้างอิงและดึงไปใช้งานร่วมกันได้:

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

package sys_config_pkg is
    constant C_CLK_FREQ  : integer := 100_000_000; -- 100 MHz
    type T_BUS_RECORD is record
        addr  : std_logic_vector(15 downto 0);
        data  : std_logic_vector(31 downto 0);
        valid : std_logic;
    end record;
end package sys_config_pkg;

5.2 Functions (function)

บล็อกการคำนวณแบบ Pure Combinational Logic ที่คืนค่ากลับมา 1 ค่าเสมอ (ไม่มี Propagation Delay ในการจำลอง และไม่สร้าง Clock Edge) นิยมใช้ทำ Math Calculation หรือหาขนาด Bit Width:

function calc_parity (val : std_logic_vector) return std_logic is
    variable v_parity : std_logic := '0';
begin
    for i in val'range loop
        v_parity := v_parity xor val(i);
    end loop;
    return v_parity;
end function calc_parity;

5.3 Procedures (procedure)

บล็อกโค้ดย่อยที่สามารถรับและส่งค่าผ่านพอร์ต in, out, inout ได้หลายค่าพร้อมกัน นิยมนำมาใช้สร้าง Helper Subprograms สำหรับไดรฟ์สัญญาณในไฟล์ Testbench ให้เขียนได้สั้นและอ่านง่ายขึ้น (สามารถใช้คำสั่ง wait for หน่วงเวลาภายใน Procedure ได้เมื่อใช้ใน Testbench)

-- 1. ประกาศและนิยาม Procedure สำหรับส่งข้อมูลผ่าน Bus ใน Testbench
procedure write_bus (
    constant addr   : in  std_logic_vector(7 downto 0);
    constant data   : in  std_logic_vector(7 downto 0);
    signal   o_addr : out std_logic_vector(7 downto 0);
    signal   o_data : out std_logic_vector(7 downto 0);
    signal   o_valid: out std_logic
) is
begin
    o_addr  <= addr;
    o_data  <= data;
    o_valid <= '1';
    wait for 10 ns; -- หน่วงเวลา 1 Clock Cycle
    o_valid <= '0';
    wait for 10 ns;
end procedure write_bus;

-- 2. การเรียกใช้งานภายใน Stimulus Process ของ Testbench
p_stimulus : process
begin
    -- ส่งข้อมูลผ่าน Procedure ได้สั้นกระชับ ไม่ต้องเขียนโค้ดซ้ำหลายบรรทัด
    write_bus(X"0A", X"FF", s_tb_addr, s_tb_data, s_tb_valid);
    write_bus(X"0B", X"A5", s_tb_addr, s_tb_data, s_tb_valid);
    wait;
end process p_stimulus;

6 Signal vs Variable

สิ่งสำคัญที่สุดในการเขียน VHDL คือความเข้าใจเรื่อง การอัปเดตค่าของ Signal และ Variable:

คุณสมบัติ (Feature) Signal (signal) Variable (variable)
ขอบเขตการใช้งาน (Scope) ประกาศภายนอก Process (ใช้เชื่อมต่อได้ทั้ง Architecture) ประกาศและใช้ได้ ภายใน Process เท่านั้น
ตัวดำเนินการกำหนดค่า <= (Signal Assignment) := (Variable Assignment)
ช่วงเวลาการอัปเดตค่า Scheduled Update: ค่าจะยังไม่อัปเดตทันที แต่จะอัปเดตเมื่อจบ Process Delta Cycle Immediate Update: ค่าจะเปลี่ยนแปลง ทันที ในบรรทัดนั้นๆ
ผลลัพธ์ฮาร์ดแวร์ สร้างเป็นสายสัญญาณ (Wire) หรือ Flip-Flop มักกลายเป็นสายสัญญาณ Combinational Intermediate หรือ Register ขึ้นอยู่กับโครงสร้าง
ข้อควรระวัง Signal vs Variable:

การใช้ Variable กำหนดค่าหลายครั้งใน Process เดียวกัน จะสร้าง Combinational Logic ที่ต่อซ้อนกันเป็นทอดๆ (Cascaded Logic) ซึ่งอาจส่งผลเสียต่อ Data Path Delay และเกิด Timing Violation ได้ หากไม่จำเป็นควรเน้นใช้ signal เป็นหลัก

7 Signal Naming example

เพื่อเพิ่มความอ่านง่าย (Readability) และลดความผิดพลาดในการเชื่อมต่อโมดูลความซับซ้อนสูง การออกแบบ FPGA มักมีข้อตกลงในการเติม Prefix/Suffix ให้กับสัญญาณ

ข้อควรทราบก่อนอ่านตาราง

แต่ละบริษัท/ทีมออกแบบมี Naming Convention ของตัวเอง ซึ่งไม่มีสูตรตายตัวว่าแบบไหนถูกหรือผิด สิ่งที่สำคัญที่สุดคือ เลือกใช้ Convention เดียวให้สม่ำเสมอตลอดทั้งโปรเจกต์ ห้ามผสมหลาย Convention ปนกัน (เช่น ใช้ทั้ง _n และ _l แทนความหมาย Active-Low ในโปรเจกต์เดียวกัน) เพราะจะทำให้ทีมสับสนและอ่านโค้ดยากขึ้น
หัวข้อ 8.1 เป็นตัวอย่าง Convention หลักที่แนะนำ ส่วน 8.2 เป็น Convention อื่นๆ ที่พบได้บ่อยในอุตสาหกรรม ไว้สำหรับอ้างอิงเวลาไปอ่านโค้ดของทีม/บริษัทอื่น

8 Synthesizable vs Non-synthesizable (Testbench-only) VHDL Syntax

ภาษา VHDL ถูกออกแบบมาให้ใช้ได้ทั้ง สร้างฮาร์ดแวร์จริง (Synthesis) และ จำลองพฤติกรรม (Simulation) แต่ไม่ใช่ทุกคำสั่งใน VHDL จะสามารถ "สังเคราะห์" ออกมาเป็นวงจรจริงบนชิปได้ บาง Syntax มีไว้สำหรับ Testbench เท่านั้น (Simulation-only) เพราะมันอธิบายพฤติกรรมของ "เวลา" หรือ "สภาพแวดล้อมการทดสอบ" ซึ่งไม่มีตัวตนอยู่จริงในฮาร์ดแวร์

💡 หลักการ
  • Synthesizable Code (RTL) = โค้ดที่ Synthesis Tool (Quartus, Vivado) แปลงเป็น Gate/Flip-Flop/Wire จริงได้
  • Non-synthesizable Code (Testbench-only) = โค้ดที่ใช้ได้เฉพาะตอน Simulate (ModelSim, QuestaSim, GHDL) เพื่อสร้าง Stimulus หรือตรวจสอบผลลัพธ์ ถ้านำไปสังเคราะห์ Synthesis Tool จะ Error หรือเพิกเฉย (Ignore) ทันที

ผู้เริ่มต้นศึกษา FPGA ส่วนใหญ่มักเจอปัญหานี้เป็นด่านแรกๆ คือ แยกไม่ออกว่า Syntax ไหน "Synthesizable" (สังเคราะห์เป็นวงจรจริงได้) และ Syntax ไหน "Un-synthesizable" (ใช้ได้แค่ตอน Simulate) เพราะ Compiler ของ Simulator (เช่น ModelSim) จะ Compile ผ่านหมดทุกคำสั่ง โดยไม่มี Error ใดๆ ทำให้เข้าใจผิดว่าโค้ดที่รันผ่านใน Simulation จะสังเคราะห์เป็นฮาร์ดแวร์ได้เสมอ ทั้งที่จริงแล้ว VHDL เป็น Superset ที่ใหญ่กว่า Hardware ที่สังเคราะห์ได้จริงมาก ความสับสนนี้เป็นสาเหตุอันดับต้นๆ ที่ทำให้ Code รันบน Simulator ได้ปกติ แต่พอนำไป Synthesis กลับ Error หรือได้วงจรที่ทำงานผิดจากที่ตั้งใจ ตารางด้านล่างนี้จึงสรุป Keyword/Syntax ที่พบบ่อยไว้ให้

8.1 ตารางเปรียบเทียบ Syntax ที่พบบ่อย

Syntax / Construct Synthesizable (RTL) Non-synthesizable (Testbench-only) หมายเหตุ
process(clk, rst_n) + rising_edge(clk) ✅ ได้ ✅ ได้ รูปแบบมาตรฐานของ Sequential Logic ใช้ได้ทั้งสองฝั่ง
process(all) / process(input_list) ✅ ได้ ✅ ได้ Combinational Logic ใช้ได้ทั้งสองฝั่ง
wait for 10 ns; ❌ ไม่ได้ ✅ ได้ หน่วงเวลาแบบ "สัมบูรณ์" ไม่มีอยู่จริงในฮาร์ดแวร์ (ฮาร์ดแวร์รอได้แค่ตาม Clock Edge เท่านั้น)
wait until <condition>; ❌ ไม่ได้ ✅ ได้ ใช้รอเงื่อนไขใน Stimulus Process เท่านั้น
wait; (ไม่มีเงื่อนไข) ❌ ไม่ได้ ✅ ได้ หยุด Process ค้างไว้ตลอดไป ใช้ปิดท้าย Stimulus Process
signal <= value after 10 ns; ⚠️ ส่วนใหญ่ไม่ได้ ✅ ได้ after clause (Transport/Inertial Delay) ถูก เพิกเฉย โดย Synthesis Tool ส่วนใหญ่
assert <cond> report "msg" severity ...; ⚠️ Tool-dependent ✅ ได้ บาง Tool รองรับเป็น Synthesis Assertion แต่ส่วนใหญ่ใช้เฉพาะ Debug ใน Simulation
report "message"; ❌ ไม่ได้ ✅ ได้ พิมพ์ข้อความออก Console ของ Simulator เท่านั้น
std.textio (file_open, readline, ...) ❌ ไม่ได้ ✅ ได้ อ่าน/เขียนไฟล์บน Disk — ไม่มีแนวคิด "ไฟล์" อยู่ในฮาร์ดแวร์
std.env.stop / std.env.finish ❌ ไม่ได้ ✅ ได้ สั่งจบการ Simulation (VHDL-2008)
integer (มีขอบเขต / Constrained Range) ✅ ได้ ✅ ได้ ต้องระบุ Range ให้ชัดเจน (เช่น integer range 0 to 255) เพื่อให้สังเคราะห์เป็น Bit-width ที่แน่นอน
for-loop แบบมีขอบเขตตายตัว (Static Range) ✅ ได้ (Unroll) ✅ ได้ สังเคราะห์โดยการ "คลี่ลูป" ตามจำนวนรอบที่กำหนดไว้ตายตัว
while-loop / Loop ที่ขอบเขตไม่แน่นอน ❌ ไม่ได้ ✅ ได้ Synthesis Tool ไม่สามารถคลี่ลูปที่ไม่รู้จำนวนรอบล่วงหน้าได้
std_logic ค่า 'U', 'W', 'L', 'H', '-' ⚠️ มีความหมายจำกัด ✅ ได้ครบ 9 ค่า ฮาร์ดแวร์จริงมีแค่ '0', '1', 'Z' เท่านั้น ค่าอื่นมีไว้ช่วย Debug ใน Simulation

8.2 ตัวอย่างโค้ด: รูปแบบ Non-synthesizable (Testbench-only) ที่พบบ่อย

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use STD.TEXTIO.ALL;

entity tb_my_module is
end entity tb_my_module; -- Testbench ไม่มี Port

architecture sim of tb_my_module is
    constant C_CLK_PERIOD : time := 10 ns; -- 'time' type: Non-synthesizable

    signal clk    : std_logic := '0';
    signal rst_n  : std_logic := '0';
    signal i_data : std_logic_vector(7 downto 0) := (others => '0');
    signal o_data : std_logic_vector(7 downto 0);
begin

    -- Instance ของวงจรจริง (DUT: Design Under Test)
    u_dut : entity work.my_module
        port map (
            clk    => clk,
            rst_n  => rst_n,
            i_data => i_data,
            o_data => o_data
        );

    ------------------------------------------------------------------
    -- Clock Generator: ใช้ wait for สร้างสัญญาณนาฬิกา (Non-synthesizable)
    ------------------------------------------------------------------
    p_clk_gen : process
    begin
        clk <= '0';
        wait for C_CLK_PERIOD / 2;
        clk <= '1';
        wait for C_CLK_PERIOD / 2;
    end process p_clk_gen;

    ------------------------------------------------------------------
    -- Stimulus Process: สั่งค่าอินพุตตามลำดับเวลา แล้วปิดท้ายด้วย wait;
    ------------------------------------------------------------------
    p_stimulus : process
    begin
        rst_n  <= '0';
        wait for 20 ns;              -- absolute delay: Non-synthesizable
        rst_n  <= '1';

        wait until rising_edge(clk); -- รอ Edge: Non-synthesizable
        i_data <= X"AA";

        wait for 40 ns;
        assert (o_data = X"AA")      -- assert/report: Non-synthesizable
            report "o_data mismatch!" severity error;

        wait; -- หยุด Process ค้างตลอดไป จบการทดสอบ
    end process p_stimulus;

end architecture sim;
Syntax เหล่านี้ไม่สามารถสังเคราะห์เป็นวงจรจริงได้
  • wait for, wait until, assert/report, std.textio, ชนิดข้อมูล time/real
  • Clock Generator (p_clk_gen แบบข้างต้น)

8.3 วิธีเช็คว่าโค้ดของเรา Synthesizable หรือ Non-synthesizable
